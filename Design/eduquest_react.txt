// --- 元件：主選單關卡卡片 ---
const LevelCard = ({ level, onSelect }) => {
  const isLocked = level.status === 'locked';
  
  return (
    <div className={`
      relative group overflow-hidden rounded-2xl border-2 transition-all duration-300
      ${isLocked ? 'bg-gray-50 border-gray-100 opacity-80' : 'bg-white border-transparent hover:border-blue-200 hover:shadow-xl'}
    `}>
      {/* 背景裝飾 */}
      <div className={`absolute top-0 right-0 p-8 opacity-5 group-hover:scale-110 transition-transform ${isLocked ? 'text-gray-400' : 'text-blue-600'}`}>
        {level.icon}
      </div>

      <div className="p-8 relative">
        <div className="flex justify-between items-start mb-6">
          <span className={`px-3 py-1 rounded-full text-xs font-bold uppercase tracking-wider ${
            isLocked ? 'bg-gray-200 text-gray-500' : 'bg-blue-100 text-blue-600'
          }`}>
            關卡 {level.id}
          </span>
          {level.progress > 0 && (
            <span className="text-sm font-semibold text-blue-600">{level.progress}%</span>
          )}
        </div>

        <h3 className="text-2xl font-bold text-gray-900 mb-3">{level.title}</h3>
        <p className="text-gray-600 text-sm leading-relaxed mb-8 line-clamp-2">
          {level.description}
        </p>

        {/* 進度條 */}
        <div className="w-full h-1.5 bg-gray-100 rounded-full mb-8 overflow-hidden">
          <div 
            className={`h-full transition-all duration-500 ${level.progress === 100 ? 'bg-green-500' : 'bg-blue-500'}`}
            style={{ width: `${level.progress}%` }}
          />
        </div>

        <button
          disabled={isLocked}
          onClick={() => onSelect(level)}
          className={`
            w-full py-3 rounded-xl font-bold transition-all flex items-center justify-center gap-2
            ${isLocked 
              ? 'bg-gray-200 text-gray-400 cursor-not-allowed' 
              : 'bg-blue-600 text-white hover:bg-blue-700 shadow-md hover:shadow-blue-200'}
          `}
        >
          {isLocked ? '尚未解鎖' : (level.progress === 100 ? '重新複習' : '繼續學習')}
          {!isLocked && <Play size={18} fill="currentColor" />}
        </button>
      </div>
    </div>
  );
};

// --- 主應用程式元件 ---
export default function EduQuestApp() {
  const [currentView, setCurrentView] = useState('menu'); // 'menu' or 'game'
  const [selectedLevel, setSelectedLevel] = useState(null);
  const [isSubmitting, setIsSubmitting] = useState(false);

  const handleSelectLevel = (level) => {
    setSelectedLevel(level);
    setCurrentView('game');
    window.scrollTo(0, 0);
  };

  const handleBackToMenu = () => {
    setCurrentView('menu');
    setSelectedLevel(null);
  };

  const handleSubmit = () => {
    setIsSubmitting(true);
    setTimeout(() => {
      setIsSubmitting(false);
      alert('已提交解答！正在評分中...');

    <div className="min-h-screen bg-slate-50 font-sans text-slate-900">

      <main className="max-w-7xl mx-auto px-4 md:px-8 py-8 lg:py-12">
        <div className="flex flex-col lg:flex-row gap-8">
          
          {/* A. 遊戲區 (左側) */}

                    px-8 py-4 rounded-2xl font-bold flex items-center gap-2 transition-all active:scale-95
                    ${isSubmitting 
                      ? 'bg-slate-200 text-slate-400 cursor-not-allowed' 
                      : 'bg-blue-600 text-white hover:bg-blue-700 shadow-lg shadow-blue-100 hover:shadow-blue-200'}

                >
